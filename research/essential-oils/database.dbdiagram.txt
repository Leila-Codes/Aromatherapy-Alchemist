Table oils {
  id serial [primary key]
  name varchar [not null]
}

Table constituents {
  id serial [primary key]
  name varchar [not null]
}

Table oil_content {
  constituent_id serial [not null]
  oil_id serial [not null]
  concentration float [not null]
}

Table effect_categories {
  category_id serial [primary key]
  effect_type text [not null]
  category varchar [not null]
}

Table category_scores {
  category_id serial [not null]
  constituent_id serial [not null]
  score integer [not null]
}

Ref: oil_content.constituent_id > constituents.id
Ref: oil_content.oil_id > oils.id

Ref: category_scores.category_id - effect_categories.category_id
Ref: category_scores.constituent_id > constituents.id
